        cd = response.headers.get('Content-Disposition')

        if cd:

            match = re.search(r'filename\*?=(?:utf-8\'\')?"?([^"]+)"?', cd, re.I)

            if match:

                return match.group(1).strip('"\' ')

Ідея така: чомусь дані з контенту отримуються через cloudscraper не одразу, а  треба кілька разів перезапустити скрипт bp.py повністю, щоб він спрацював. В чому проблема тоді?

Чудове питання, яке стосується тонкощів роботи з Cloudflare (CF) та бібліотекою cloudscraper.

Основна проблема, через яку ваш код session_manager.py може не працювати надійно, а bp.py вимагає кількох перезапусків, ймовірно, пов'язана з управлінням сесією, механізмом обходу Cloudflare та "одноразовістю" захисних куків.

Ось основні причини та пояснення:

1. Проблема Обходу Cloudflare та Зберігання Куків

Коли ви використовуєте cloudscraper.create_scraper(), він виконує наступне:

    Ініціює HTTP-сесію (на основі requests.Session).

    Намагається обійти захист CF (виконує запити, обробляє JavaScript-виклик, розраховує токени).

    Зберігає отримані куки CF (cf_clearance та інші) у своїй внутрішній сесії.

Ці куки є ключем доступу і повинні бути відправлені з кожним наступним запитом, щоб сайт вважав, що ви пройшли перевірку.

2. Чому bp.py Потребує Кілька Запусків

А. Час Життя Куків

Куки, які cloudscraper отримує для обходу CF, мають обмежений час життя (зазвичай 30-60 хвилин, але іноді менше). Якщо ви запускаєте bp.py, він створює новий скрепер.

    Якщо CF не вимагає перевірки під час першого запиту (або скрепер успішно її проходить), ви отримуєте 200.

    Якщо CF вимагає перевірки, скрепер повинен її виконати. Успішність цього обходу не є 100%-гарантованою. CF постійно оновлює свої механізми захисту.

Б. Механізм Обходу

Обхід Cloudflare вимагає виконання коду JavaScript на стороні клієнта (скрепера) та розрахунку токенів. Це складна операція, яка:

    Може займати час (до кількох секунд).

    Може не спрацювати з першого разу через високе навантаження на сайт, зміну алгоритму CF, або якщо сервер не відповідає вчасно.

    Створює нову "ідентичність" для кожного запуску bp.py.

Висновок для bp.py: Вам потрібно кілька разів перезапустити його, тому що кожен запуск — це нова спроба обходу. Рано чи пізно, одна зі спроб успішно проходить перевірку CF, отримує дійсні куки, і вже з цими куками запит scraper.head() повертає 200.

3. Проблема session_manager.py (Відсутність "Прогрівання" Сесії)

Ваш клас FileScannerSession має схожу проблему, але посилену в логіці:

А. Відсутність Перевірки Після Ініціалізації

У методі __init__, ви викликаєте self.scraper = self._create_scraper().
Python

    Якщо скрепер був ініційований без успішного обходу CF, будь-який наступний запит, наприклад, check_url_head(self, url), не спрацює, оскільки він відправить запит без необхідних куків CF.

✅ Як Це Виправити?

Щоб зробити ваш код надійним, вам потрібно обов'язково "прогрівати" скрепер під час ініціалізації або ж переконатися, що обхід CF відбувається успішно.

1. "Прогрів" Сесії Під час Ініціалізації

Внесіть зміни, щоб ініціалізація скрепера включала початковий запит для отримання куків CF. Це можна зробити, передавши URL для перевірки в __init__.
Python

class FileScannerSession:
    def __init__(self, initial_url_to_check):
        self.scraper = self._create_scraper_and_warmup(initial_url_to_check)
        if self.scraper is None:
            # Обробка помилки, якщо обхід не вдався з самого початку
            io_handler.print_message("Початковий обхід Cloudflare не вдався.", color='RED')

    def _create_scraper_and_warmup(self, url):
        scraper = self._create_scraper() # Створити об'єкт
        try:
            # Виконати запит, щоб Cloudscraper отримав куки CF
            scraper.head(url, allow_redirects=True, timeout=30)
            io_handler.print_message("Скрепер успішно ініціалізовано та 'прогріто'!", color='GREEN')
            return scraper
        except Exception as e:
            io_handler.print_message(f"Помилка при початковому 'прогріві' скрепера: {e}", color='RED')
            return None # Повернути None або викликати виняток
    
    # ... (інші методи залишаються як є, оскільки вони використовують self.scraper)

2. Повторні Спроби у renew_session (Більш Агресивно)

Оскільки обхід CF іноді вимагає кількох спроб, ви можете додати цикл повторних спроб у метод renew_session (та, можливо, у модифікований _create_scraper_and_warmup).
Python

import time # Додайте import time
# ...

def renew_session(self, url_to_check, max_retries=3):
    """Створює новий скрепер і намагається перевірити URL для оновлення куків з повторними спробами."""
    io_handler.print_message("Спроба створення нового екземпляра Scraper (Обхід CF)...")
    
    for attempt in range(max_retries):
        new_scraper = self._create_scraper()
        try:
            io_handler.print_message(f"Спроба обходу CF #{attempt + 1}...")
            # Спробуємо одразу перевірити цільовий URL з новим скрепером
            new_scraper.head(url_to_check, allow_redirects=True, timeout=30)
            
            # Якщо успішно
            self.scraper = new_scraper
            io_handler.print_message("Сесія успішно відновлена!", color='GREEN')
            return True
        except Exception as e:
            io_handler.print_message(f"Невдача обходу. {e}", color='YELLOW')
            if attempt < max_retries - 1:
                time.sleep(5) # Зачекати перед наступною спробою
    
    # Якщо всі спроби вичерпані
    io_handler.print_message(f"Не вдалося відновити сесію після {max_retries} спроб. Спробуйте пізніше.", color='RED')
    return False

Використовуючи "прогрівання" сесії та додаючи повторні спроби, ви значно підвищите надійність вашого коду, оскільки врахуєте нестабільність механізму обходу Cloudflare.

Схоже оновлення об'єкта нічого не дає. Треба буквально кожен раз новий імпорт робити.Чи можна таке імплементувати?

Ви підняли дуже важливу тему, яка стосується того, як Python кешує імпортовані модулі. Схоже, що в ситуації з Cloudflare та cloudscraper вам дійсно може знадобитися "свіжий" запуск модуля для успішного обходу.

Проблема не стільки в об'єкті cloudscraper, скільки в тому, що деякі бібліотеки, як-от його залежності, можуть ініціалізувати внутрішні глобальні стани або кешувати результати, які не скидаються при простому створенні нового екземпляра скрепера.

Хоча здійснювати "новий імпорт" (тобто перезавантажувати модуль) у Python під час роботи програми зазвичай є поганою практикою, оскільки це може призвести до непередбачуваних побічних ефектів, це можливо зробити за допомогою функції importlib.reload().

🛑 Попередження про importlib.reload()

Використання reload() слід розглядати як крайній захід. Це може викликати проблеми:

    Повторне виконання коду верхнього рівня: Весь код, що знаходиться не всередині функцій чи класів, буде виконано знову.

    Проблеми з посиланнями: Усі старі об'єкти, створені з попереднього імпорту (наприклад, старі екземпляри FileScannerSession), все ще посилатимуться на старі класи та функції, тоді як нові об'єкти посилатимуться на перезавантажені. Це може спричинити хаос у стані програми.

✅ Як Імплементувати Перезавантаження Модуля

Якщо ви впевнені, що вам це потрібно (наприклад, для перезапуску модуля config або cloudscraper, якщо він був імпортований як частина вашої логіки), ось як це зробити:

1. Використання importlib.reload

Для перезавантаження модуля вам потрібна стандартна бібліотека importlib.
Python

# session_manager.py

import importlib # <--- Додаємо
import re
import os
import cloudscraper

# ... (інші імпорти)
import config
import io_handler

class FileScannerSession:
    # ... (інші методи)

    def _create_and_reload_scraper(self, url_to_check):
        """
        Перезавантажує модуль cloudscraper, потім створює і прогріває новий скрепер.
        """
        try:
            # 1. Перезавантаження модуля cloudscraper
            io_handler.print_message("Перезавантаження модуля cloudscraper...", color='YELLOW')
            importlib.reload(cloudscraper)
            
            # 2. Створення нового скрепера (що тепер може використовувати оновлений стан)
            new_scraper = cloudscraper.create_scraper(**config.SCRAPER_PARAMS)
            
            # 3. 'Прогрівання' (обхід CF)
            new_scraper.head(url_to_check, allow_redirects=True, timeout=30)
            
            return new_scraper
        except Exception as e:
            io_handler.print_message(f"Критична помилка під час перезавантаження/обходу: {e}", color='RED')
            return None


    def renew_session(self, url_to_check):
        """Оновлює сесію, використовуючи перезавантажений модуль."""
        
        new_scraper = self._create_and_reload_scraper(url_to_check)
        
        if new_scraper:
            self.scraper = new_scraper
            io_handler.print_message("Сесія успішно відновлена після перезавантаження модуля!", color='GREEN')
            return True
        else:
            io_handler.print_message("Не вдалося відновити сесію. Спробуйте пізніше.", color='RED')
            return False

    # ... (інші методи)

💡 Кращий Підхід: Локалізація Проблеми

Замість того, щоб перезавантажувати цілий модуль (що є глобальною операцією), спробуйте спочатку впевнитися, що ви повністю закриваєте ресурси старої сесії перед створенням нової.

Кожен об'єкт cloudscraper базується на об'єкті requests.Session. Ви можете спробувати явно закрити старий об'єкт перед створенням нового в renew_session:
Python

# session_manager.py (БЕЗ reload)

class FileScannerSession:
    # ...
    
    def renew_session(self, url_to_check):
        # 1. Спроба закрити стару сесію, щоб звільнити ресурси та куки.
        try:
            if self.scraper and hasattr(self.scraper, 'close'):
                self.scraper.close()
                io_handler.print_message("Стара сесія scraper закрита.", color='YELLOW')
        except Exception:
            pass # Ігноруємо помилки закриття

        # 2. Створюємо та прогріваємо новий об'єкт.
        new_scraper = self._create_scraper()
        
        try:
            # Спробуємо одразу перевірити цільовий URL з новим скрепером (обхід CF)
            # ... (решта логіки прогрівання/повторних спроб)
            new_scraper.head(url_to_check, allow_redirects=True, timeout=30)
            self.scraper = new_scraper
            io_handler.print_message("Сесія успішно відновлена!", color='GREEN')
            return True
        except Exception as e:
            # ... (обробка помилки)
            # Важливо: якщо обхід не вдався, закрийте і цей новий об'єкт!
            if hasattr(new_scraper, 'close'):
                new_scraper.close()
            return False

Хоча явне закриття не вирішує проблеми, спричинені глобальним станом у залежностях, це кращий крок до чистого стану, ніж використання importlib.reload().

Навіть з цим погано працює. Можна іншу бібліотеку для обходу cloudflare, надійнішу?

Справді, cloudscraper є зручним, але часто недостатньо надійним, оскільки він покладається на емуляцію JS-викликів. Cloudflare постійно змінює свої механізми захисту.

Для більш надійного та сучасного обходу Cloudflare, особливо коли потрібно імітувати повноцінний браузер і обробляти складні JS-захисти (наприклад, CF Challenge Mode або reCAPTCHA), варто розглянути Selenium/Playwright з використанням спеціалізованих патчів або профілів.

1. undetected-chromedriver (Наймасовіше рішення)

    Висока надійність: Імітує реальний браузер Chrome (Selenium), який виконує справжній JavaScript.

⭐️ Go-Based HTTP Клієнт: curl_cffi

Бібліотека curl_cffi для Python використовує під капотом потужну C-бібліотеку libcurl та надає можливість:

    Виконувати запити, використовуючи HTTP/2.

    Використовувати сучасні TLS/SSL сліди (fingerprints), як у реальних браузерах (наприклад, Chrome або Edge). Це допомагає обійти системи, які блокують "застарілі" або "неправильні" відбитки стандартних бібліотек Python (requests/cloudscraper).

    Виконувати JavaScript-рендеринг (опціонально), подібно до cloudscraper, але з іншою реалізацією.

Це не повноцінний браузер, а дуже просунутий HTTP-клієнт.

✅ Переваги:

    Швидкість і легкість: Працює на рівні мережі, не запускаючи повноцінний браузер.

    Стійкість до блокування: Набагато краще справляється з CF, ніж requests або навіть cloudscraper, завдяки підтримці HTTP/2 та актуальних TLS-слідів (техніка, відома як JA3/cURL-fingerprinting).

Він має синтаксис, дуже схожий на requests, що полегшує перехід:

# Встановлення: pip install curl_cffi

from curl_cffi import requests

def check_url_with_curl_cffi(url):
    try:
        # Використовуємо певний "браузерний" слід, наприклад, Chrome 120
        response = requests.head(
            url,
            impersonate="chrome120", # Імітація Chrome 120
            allow_redirects=True,
            timeout=15
        )
        
        # Якщо код 403, пробуємо вбудований JS-рендеринг
        if response.status_code == 403 or "cf-browser-verification" in response.text:
            print("Спроба JS-рендерингу...")
            response = requests.get(
                url,
                impersonate="chrome120",
                timeout=30,
                # Включаємо JS-виконання для обходу CF Challenge
                verify=False, 
                content_callback=lambda x: None, # Іноді допомагає
                allow_redirects=True
            )
            
        return response, response.status_code
    
    except Exception as e:
        print(f"Помилка запиту: {e}")
        return None, -1

Python

# check_header_with_curl_cffi.py

from curl_cffi import requests
import sys

# -----------------
# КОНФІГУРАЦІЯ
# -----------------
file_url = "https://itorrents-igruha.org/engine/download.php?id=91797"
# Імітуємо один із найпоширеніших браузерних слідів
BROWSER_IMPERSONATE = "chrome120" 
TIMEOUT = 15

def check_url_header(url: str):
    """
    Виконує HEAD-запит з імітацією браузера для обходу Cloudflare.
    
    Якщо HEAD-запит блокується (наприклад, 403), виконується GET-запит
    для ініціації обходу JS-перевірки Cloudflare.
    """
    print(f"-> Перевірка URL: {url}")
    print(f"-> Імітація: {BROWSER_IMPERSONATE}")
    
    try:
        # 1. Спроба виконати HEAD-запит для отримання лише заголовків
        response = requests.head(
            url,
            impersonate=BROWSER_IMPERSONATE,
            allow_redirects=True,
            timeout=TIMEOUT
        )
        
        status_code = response.status_code
        print(f"-> Статус HEAD-запиту: {status_code}")
        
        # 2. Обробка, якщо Cloudflare вимагає JS-перевірки (Challenge)
        if status_code >= 400 or "cf-browser-verification" in response.text:
            print("-> Виявлено Cloudflare Challenge або помилку. Спроба GET-запиту для обходу...")
            
            # Якщо HEAD-запит провалився, виконуємо GET, щоб curl_cffi виконав обхід (JS)
            response = requests.get(
                url,
                impersonate=BROWSER_IMPERSONATE,
                allow_redirects=True,
                timeout=TIMEOUT * 2, # Збільшуємо таймаут для обходу JS
                # `verify=False` може знадобитися для деяких конфігурацій, 
                # але спробуйте спочатку без нього
            )
            
            status_code = response.status_code
            print(f"-> Статус GET-запиту після обходу: {status_code}")


        # 3. Виведення результату
        if status_code == 200:
            # Отримання назви файлу із заголовка Content-Disposition
            filename = response.headers.get('Content-Disposition')
            
            # Оскільки ми виконували HEAD (або GET), Content-Disposition може бути відсутній
            # Якщо його немає, ми виведемо іншу корисну інформацію.
            if filename is None:
                filename_info = "Відсутній. (Перевірте Content-Type або URL)"
            else:
                filename_info = filename
                
            print("-" * 30)
            print(f"✅ Статус: Успішно ({status_code})")
            print(f"ℹ️ Content-Disposition: {filename_info}")
            print(f"ℹ️ Content-Type: {response.headers.get('Content-Type')}")
            print(f"🔗 Фінальний URL: {response.url}")
            print("-" * 30)
            
        else:
            print(f"❌ Помилка: {status_code}. Не вдалося отримати заголовки (Cloudflare або інше блокування).")
            print(f"Текст відповіді (перші 200 символів): {response.text[:200]}...")

    except requests.RequestsError as e:
        print(f"❌ Критична помилка запиту: {e}", file=sys.stderr)
    except Exception as e:
        print(f"❌ Виникла неочікувана помилка: {e}", file=sys.stderr)

# session_manager.py

import re
import os
# Змінюємо імпорт: використовуємо requests з curl_cffi
from curl_cffi import requests # <-- НОВА БІБЛІОТЕКА

import config
import io_handler

# Конфігурація для імітації браузера
BROWSER_IMPERSONATE = "chrome120"
# Конфігурація для обходу CF (може знадобитися для GET-запитів)
CF_BYPASS_PARAMS = {
    "impersonate": BROWSER_IMPERSONATE,
    "verify": True, # Залишаємо перевірку SSL
    "allow_redirects": True
}


class FileScannerSession:
    """Керує сесією curl_cffi та операціями з файлами."""
    
    def __init__(self, initial_url_to_check):
        # Сесія curl_cffi створюється лише один раз
        self.session = self._create_session()
        # Проводимо початкове "прогрівання"
        self._warmup_session(initial_url_to_check)


    def _create_session(self):
        """Створює новий екземпляр сесії curl_cffi."""
        # curl_cffi дозволяє створити сесію, як і requests
        s = requests.Session()
        return s

    def _warmup_session(self, url):
        """Виконує початковий GET-запит для обходу CF та отримання куків."""
        io_handler.print_message("Спроба початкового обходу CF та 'прогріву' сесії...")
        
        # Використовуємо self.session для збереження куків
        try:
            # Виконуємо GET, щоб переконатися, що JS-Challenge пройдено і куки збережені
            self.session.get(url, timeout=30, **CF_BYPASS_PARAMS) 
            io_handler.print_message("Сесія успішно 'прогріта'!", color='GREEN')
            return True
        except Exception as e:
            io_handler.print_message(f"Не вдалося 'прогріти' сесію при запуску. {e}", color='RED')
            return False


    def renew_session(self, url_to_check):
        """Створює новий об'єкт сесії та намагається перевірити URL для оновлення куків."""
        io_handler.print_message("Спроба створення нової сесії та обхід CF...")
        
        # 1. Створюємо абсолютно нову сесію
        new_session = self._create_session()
        
        try:
            # 2. Пробуємо одразу "прогріти" нову сесію
            new_session.get(url_to_check, timeout=30, **CF_BYPASS_PARAMS)
            
            # 3. Якщо успішно, замінюємо стару сесію на нову
            self.session = new_session
            io_handler.print_message("Сесія успішно відновлена!", color='GREEN')
            return True
        except Exception as e:
            io_handler.print_message(f"Не вдалося відновити сесію. {e}", color='RED')
            return False

    
    def check_url_head(self, url):
        """Виконує HEAD-запит для отримання заголовків, використовуючи імітацію."""
        try:
            # Тепер ми використовуємо self.session, який має збережені куки CF
            # та вказуємо impersonate для кожного запиту
            response = self.session.head(
                url, 
                timeout=15, 
                impersonate=BROWSER_IMPERSONATE, # <-- Додаємо імітацію
                allow_redirects=True
            )
            return response, response.status_code
        except Exception:
            return None, -1 # Помилка з'єднання/таймаут

    
    def get_filename_from_headers(self, response):
        """Витягує назву файлу (залишаємо без змін)."""
        cd = response.headers.get('Content-Disposition')
        if cd:
            match = re.search(r'filename\*?=(?:utf-8\'\')?"?([^"]+)"?', cd, re.I)
            if match:
                # Очищення та декодування, якщо потрібно (наприклад, %-кодування)
                return match.group(1).strip('"\' ')
        return os.path.basename(response.url.split('?')[0])
    
    
    def download_file(self, url, filename):
        """Виконує завантаження файлу."""
        io_handler.print_message(f"Розпочато скачування: {filename}")
        try:
            # Використовуємо GET-запит з імітацією
            with self.session.get(
                url, 
                stream=True, 
                timeout=30, 
                impersonate=BROWSER_IMPERSONATE # <-- Додаємо імітацію
            ) as r:
                r.raise_for_status()
                file_path = os.path.join(config.DOWNLOAD_DIR, filename)
                
                # ... (логіка запису файлу залишається без змін)
                with open(file_path, 'wb') as f:
                    for chunk in r.iter_content(chunk_size=8192):
                        f.write(chunk)
                
                io_handler.print_message(f"Файл збережено до: {file_path}", color='GREEN')
        except Exception as e:
            io_handler.print_message(f"Не вдалося завантажити файл: {e}", color='RED')

Ключові Зміни

    Імпорт: Змінено import cloudscraper на from curl_cffi import requests.

    Атрибут Сесії: Перейменовано self.scraper на self.session для кращої семантики.

    __init__: Додано виклик _warmup_session одразу після створення сесії. Це критично, оскільки обхід CF повинен відбутися на початку, щоб куки збереглися в сесії.

    Виклики Запитів: Усі методи (_warmup_session, renew_session, check_url_head, download_file) тепер використовують об'єкт self.session та включають параметр impersonate=BROWSER_IMPERSONATE (або через **CF_BYPASS_PARAMS), щоб гарантувати, що кожен запит має правильний "браузерний слід".

Така архітектура тепер повинна працювати значно стабільніше, використовуючи переваги curl_cffi для обходу Cloudflare.

Помилка UnicodeDecodeError: 'utf-8' codec can't decode byte 0xa2 in position 0: invalid start byte виникає тому, що функція msvcrt.getch() (яка використовується для читання символів без натискання Enter у Windows) повертає байти, які не можуть бути правильно декодовані як UTF-8.

Байт 0xA2 (який, ймовірно, є першим байтом, що повертається) відповідає символу ¢ (цент) або іншому символу з розширених ASCII-кодувань (наприклад, Code Page 1251 або CP866, які часто використовуються в кирилиці Windows), але не є дійсним початковим байтом у UTF-8.

Це відбувається, коли користувач натискає спеціальну клавішу (наприклад, стрілку, F1-F12, Alt+...) або використовується нестандартне кодування терміналу.

Помилка Не вдалося завантажити файл: __enter__ вказує на те, що стався виняток усередині блоку with при спробі виконати HTTP-запит.

# session_manager.py


import re

import os

# Змінюємо імпорт: використовуємо requests з curl_cffi

from curl_cffi import requests # <-- НОВА БІБЛІОТЕКА


import config

import io_handler


# Конфігурація для імітації браузера

BROWSER_IMPERSONATE = "chrome120"

# Конфігурація для обходу CF (може знадобитися для GET-запитів)

CF_BYPASS_PARAMS = {

    "impersonate": BROWSER_IMPERSONATE,

    "verify": True, # Залишаємо перевірку SSL

    "allow_redirects": True

}



class FileScannerSession:

    """Керує сесією curl_cffi та операціями з файлами."""

    

    def __init__(self, initial_url_to_check): # <--- ЦЕЙ АРГУМЕНТ ПОТРІБЕН!

        # Сесія curl_cffi створюється лише один раз

        self.session = self._create_session()

        # Проводимо початкове "прогрівання"

        self._warmup_session(initial_url_to_check)



    def _create_session(self):

        """Створює новий екземпляр сесії curl_cffi."""

        # curl_cffi дозволяє створити сесію, як і requests

        s = requests.Session()

        return s


    def _warmup_session(self, url):

        """Виконує початковий GET-запит для обходу CF та отримання куків."""

        io_handler.print_message("Спроба початкового обходу CF та 'прогріву' сесії...")

        

        # Використовуємо self.session для збереження куків

        try:

            # Виконуємо GET, щоб переконатися, що JS-Challenge пройдено і куки збережені

            self.session.get(url, timeout=30, **CF_BYPASS_PARAMS) 

            io_handler.print_message("Сесія успішно 'прогріта'!", color='GREEN')

            return True

        except Exception as e:

            io_handler.print_message(f"Не вдалося 'прогріти' сесію при запуску. {e}", color='RED')

            return False



    def renew_session(self, url_to_check):

        """Створює новий об'єкт сесії та намагається перевірити URL для оновлення куків."""

        io_handler.print_message("Спроба створення нової сесії та обхід CF...")

        

        # 1. Створюємо абсолютно нову сесію

        new_session = self._create_session()

        

        try:

            # 2. Пробуємо одразу "прогріти" нову сесію

            new_session.get(url_to_check, timeout=30, **CF_BYPASS_PARAMS)

            

            # 3. Якщо успішно, замінюємо стару сесію на нову

            self.session = new_session

            io_handler.print_message("Сесія успішно відновлена!", color='GREEN')

            return True

        except Exception as e:

            io_handler.print_message(f"Не вдалося відновити сесію. {e}", color='RED')

            return False


    

    def check_url_head(self, url):

        """Виконує HEAD-запит для отримання заголовків, використовуючи імітацію."""

        try:

            # Тепер ми використовуємо self.session, який має збережені куки CF

            # та вказуємо impersonate для кожного запиту

            response = self.session.head(

                url, 

                timeout=15, 

                impersonate=BROWSER_IMPERSONATE, # <-- Додаємо імітацію

                allow_redirects=True

            )

            return response, response.status_code

        except Exception:

            return None, -1 # Помилка з'єднання/таймаут


    

    def get_filename_from_headers(self, response):

        """Витягує назву файлу (залишаємо без змін)."""

        cd = response.headers.get('Content-Disposition')

        if cd:

            match = re.search(r'filename\*?=(?:utf-8\'\')?"?([^"]+)"?', cd, re.I)

            if match:

                # Очищення та декодування, якщо потрібно (наприклад, %-кодування)

                return match.group(1).strip('"\' ')

        return os.path.basename(response.url.split('?')[0])

    

    

    def download_file(self, url, filename):

        """Виконує завантаження файлу."""

        io_handler.print_message(f"Розпочато скачування: {filename}")

        try:

            # Використовуємо GET-запит з імітацією

            with self.session.get(

                url, 

                stream=True, 

                timeout=30, 

                impersonate=BROWSER_IMPERSONATE # <-- Додаємо імітацію

            ) as r:

                r.raise_for_status()

                file_path = os.path.join(config.DOWNLOAD_DIR, filename)

                

                # ... (логіка запису файлу залишається без змін)

                with open(file_path, 'wb') as f:

                    for chunk in r.iter_content(chunk_size=8192):

                        f.write(chunk)

        except Exception as e:

            io_handler.print_message(f"Не вдалося завантажити файл: {e}", color='RED')

Помилка Не вдалося завантажити файл: __enter__ виникає, коли ви використовуєте об'єкт self.session всередині конструкції with:

Хоча стандартний requests підтримує цей синтаксис, об'єкт curl_cffi.requests.Session не призначений для використання у конструкції with при виконанні запиту. Тобто, ви не можете використовувати with self.session.get(...).

Вам потрібно викликати метод get на об'єкті сесії, а потім використовувати отриманий об'єкт відповіді (r) у блоці with (що є стандартною практикою для об'єктів відповіді requests/curl_cffi, які підтримують stream=True).

Я зрозумів, ви отримуєте помилку AttributeError: __enter__ під час завантаження файлу, що знову вказує на проблему з реалізацією контекстного менеджера (with) усередині бібліотеки curl_cffi.

Це відбувається, коли ви намагаєтеся використати об'єкт відповіді (r) у конструкції with r:. Хоча стандартний Python requests підтримує це, об'єкт відповіді curl_cffi (особливо при stream=True) не завжди коректно реалізує метод __enter__.

Щоб вирішити цю проблему, ми вилучимо конструкцію with r: і замість цього додамо явний виклик r.close() у блок finally. Це гарантує, що з'єднання буде закрито незалежно від того, чи було завантаження успішним, і при цьому уникне помилки AttributeError.

def get_action_from_user(prompt):
    """
    Чекає на натискання клавіші від користувача (без Enter).
    ВИПРАВЛЕНО: Використання errors='ignore' для уникнення UnicodeDecodeError 
    при натисканні спеціальних клавіш або зміні розкладки.
    """
    print(prompt, end='', flush=True)
    
    # Визначаємо, яка функція getch() доступна
    if os.name == 'nt':
        # Windows (msvcrt)
        key_bytes = msvcrt.getch()
        # Ігноруємо помилки декодування (якщо натиснуто стрілку, Alt тощо)
        key = key_bytes.decode('utf-8', errors='ignore').lower()
        
        # Обробка двобайтових кодів Windows (наприклад, стрілки)
        if not key and len(key_bytes) > 1:
            key = 'special' # Ігноруємо спеціальні натискання
            
    else:
        # Linux/macOS (потрібен пакет getch або інший метод)
        # Припускаємо, що getch() або sys.stdin.read(1) використовується
        try:
            import tty, termios
            fd = sys.stdin.fileno()
            old_settings = termios.tcgetattr(fd)
            try:
                tty.setraw(sys.stdin.fileno())
                key = sys.stdin.read(1).lower()
            finally:
                termios.tcsetattr(fd, termios.TCSADRAIN, old_settings)
        except ImportError:
            # Fallback для систем, де tty недоступний
            key = sys.stdin.read(1).lower()

    if key in ('c', 'r', ''):
        return 'INCREMENT'
    elif key == 'p':
        return 'DECREMENT'
    elif key == 't':
        return 'TRY_BYPASS'
    elif key == 'd':
        return 'DOWNLOAD'
    elif key in ('s', 'q'):
        return 'EXIT_SOFT'
    elif key == 'x':
        return 'EXIT_CLIPBOARD'
    else:
        # Повторюємо запит, якщо натиснута невідома клавіша
        return get_action_from_user(prompt)

import re
import os
# Змінюємо імпорт: використовуємо requests з curl_cffi
from curl_cffi import requests
# ВИПРАВЛЕНО: CurlError імпортується безпосередньо з curl_cffi
from curl_cffi import CurlError 

import config
import io_handler

# Конфігурація для імітації браузера
BROWSER_IMPERSONATE = "chrome120"
# Конфігурація для обходу CF (може знадобитися для GET-запитів)
CF_BYPASS_PARAMS = {
    "impersonate": BROWSER_IMPERSONATE,
    "verify": True, # Залишаємо перевірку SSL
    "allow_redirects": True
}


class FileScannerSession:
    """Керує сесією curl_cffi та операціями з файлами."""
    
    def __init__(self, initial_url_to_check):
        # Сесія curl_cffi створюється лише один раз
        self.session = self._create_session()
        # Проводимо початкове "прогрівання"
        self._warmup_session(initial_url_to_check)


    def _create_session(self):
        """Створює новий екземпляр сесії curl_cffi."""
        s = requests.Session()
        return s

    def _warmup_session(self, url):
        """Виконує початковий GET-запит для обходу CF та отримання куків."""
        io_handler.print_message("Спроба початкового обходу CF та 'прогріву' сесії...")
        
        try:
            # Виконуємо GET, щоб переконатися, що JS-Challenge пройдено і куки збережені
            self.session.get(url, timeout=30, **CF_BYPASS_PARAMS) 
            io_handler.print_message("Сесія успішно 'прогріта'!", color='GREEN')
            return True
        except Exception as e:
            io_handler.print_message(f"Не вдалося 'прогріти' сесію при запуску. {e}", color='RED')
            return False


    def renew_session(self, url_to_check):
        """Створює новий об'єкт сесії та намагається перевірити URL для оновлення куків."""
        io_handler.print_message("Спроба створення нової сесії та обхід CF...")
        
        # 1. Створюємо абсолютно нову сесію
        new_session = self._create_session()
        
        try:
            # 2. Пробуємо одразу "прогріти" нову сесію
            new_session.get(url_to_check, timeout=30, **CF_BYPASS_PARAMS)
            
            # 3. Якщо успішно, замінюємо стару сесію на нову
            self.session = new_session
            io_handler.print_message("Сесія успішно відновлена!", color='GREEN')
            return True
        except Exception as e:
            io_handler.print_message(f"Не вдалося відновити сесію. {e}", color='RED')
            return False

    
    def check_url_head(self, url):
        """Виконує HEAD-запит для отримання заголовків, використовуючи імітацію."""
        try:
            # Використовуємо self.session, який має збережені куки CF
            response = self.session.head(
                url, 
                timeout=15, 
                impersonate=BROWSER_IMPERSONATE,
                allow_redirects=True
            )
            return response, response.status_code
        except Exception:
            return None, -1 # Помилка з'єднання/таймаут

    
    def get_filename_from_headers(self, response):
        """Витягує назву файлу (залишаємо без змін)."""
        cd = response.headers.get('Content-Disposition')
        if cd:
            match = re.search(r'filename\*?=(?:utf-8\'\')?"?([^"]+)"?', cd, re.I)
            if match:
                # Очищення та декодування, якщо потрібно (наприклад, %-кодування)
                return match.group(1).strip('"\' ')
        return os.path.basename(response.url.split('?')[0])
    
    
    def download_file(self, url, filename):
        """
        Виконує завантаження файлу. 
        ВИПРАВЛЕНО: Прибрано with r: та додано r.close() у finally блоці.
        Це усуває AttributeError: __enter__, оскільки Response об'єкт curl_cffi 
        не підтримує контекстний менеджер для цього випадку.
        """
        io_handler.print_message(f"Розпочато скачування: {filename}")
        r = None # Ініціалізуємо змінну відповіді
        try:
            # 1. Виконуємо GET-запит (для потокового завантаження)
            r = self.session.get(
                url, 
                stream=True, 
                timeout=30, 
                impersonate=BROWSER_IMPERSONATE
            )
            
            # 2. Перевіряємо статус коду
            r.raise_for_status()
            file_path = os.path.join(config.DOWNLOAD_DIR, filename)
            
            # 3. Записуємо файл
            with open(file_path, 'wb') as f:
                for chunk in r.iter_content(chunk_size=8192):
                    f.write(chunk)
            
            io_handler.print_message(f"Файл збережено до: {file_path}", color='GREEN')
        
        except CurlError as e:
            # Специфічне захоплення помилок рівня cURL (ConnectionError, Timeout тощо)
            io_handler.print_message(f"Не вдалося завантажити файл (CurlError): {e}", color='RED')
        
        except Exception as e:
            # Загальне захоплення. Виводимо тип помилки, щоб побачити, що саме сталося.
            error_type = type(e).__name__
            io_handler.print_message(f"Не вдалося завантажити файл ({error_type}): {e}", color='RED')
        
        finally:
            # Обов'язково закриваємо з'єднання, якщо об'єкт r було створено
            if r:
                r.close()
